scene_start "intro"

character Alice
{
    name "Алиса"
    s mood "neutral"
    s surname "Вася"
    i relaxed 9
    b drunk false
}
character Bob
{
    name "Боб"
    i relaxed 5
    b relaxed_enough true
}
background "forest_day.jpg"
music "calm_theme.mp3"

! "Лес был наполнен мягким светом утреннего солнца."
@ "Но в нем таилась страшная тайна."

Alice "Какой сегодня прекрасный день, правда?"
Bob "Да, погода просто замечательная!"

b good_day false
s author_name "Artem"

choice
{
    "Пойти по тропинке"
    {
        ! "Вы решаете прогуляться по тропинке."
        Alice.mood "curious"
        Alice "Интересно, куда она ведёт?"
    }
    "Остаться на поляне"
    {
        ! "Вы решаете остаться на поляне и отдохнуть."
        Bob.relaxed 15
        Bob "Иногда полезно просто посидеть и послушать птиц."
    }
}

if Alice.mood == "curious" && (Bob.relaxed_enough == true || good_day == true)
{
    ! "Алиса с интересом осматривает окрестности."
    good_day true
    goto "NextScene"
}
else
{
    author_name "Victor"
    ! "Боб наслаждается спокойствием."
}

savepoint "После выбора"
scene_end

scene_start "NextScene"
background "bask/forest_path.jpg"
music "mystery_theme.mp3"
! "Вы углубляетесь в лес, где всё кажется загадочнее..."
scene_end
