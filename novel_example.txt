scene_start "intro"

character Alice
{
    name "Алиса"
    var mood "neutral"
}
character Bob
{
    name "Боб"
    relaxed 5
}
background "forest_day.jpg"
music "calm_theme.mp3"

narrate "Лес был наполнен мягким светом утреннего солнца."

Alice "Какой сегодня прекрасный день, правда?"
Bob "Да, погода просто замечательная!"

good_day false

choice
{
    "Пойти по тропинке"
    {
        narrate "Вы решаете прогуляться по тропинке."
        Alice.mood "curious"
        Alice "Интересно, куда она ведёт?"
    }
    "Остаться на поляне"
    {
        narrate "Вы решаете остаться на поляне и отдохнуть."
        Bob.relaxed 15
        Bob "Иногда полезно просто посидеть и послушать птиц."
    }
}

if Alice.mood == "curious"
{
    narrate "Алиса с интересом осматривает окрестности."
    good_day true
    goto "NextScene"
}
else
{
    narrate "Боб наслаждается спокойствием."
}

savepoint "После выбора"
scene_end

scene_start "NextScene"
background "forest_path.jpg"
music "mystery_theme.mp3"
narrate "Вы углубляетесь в лес, где всё кажется загадочнее..."
scene_end
